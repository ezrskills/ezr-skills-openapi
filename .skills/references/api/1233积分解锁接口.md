### 接口说明

#### 接口基础信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1233 |
| 接口名称 | 积分解锁 |
| 接口地址 | /api/cvip/unlockbonus |
| 协议 | HTTPS |
| 请求方式 | POST |
| 数据格式 | application/x-www-form-urlencoded |
| 字符编码 | UTF-8 |

#### 功能描述

用于解锁会员已锁定的积分，将锁定积分恢复为可用状态，适用于取消积分兑换、取消积分抵扣等场景。

## 请求结构

### 外层请求参数

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| AppId | string | 是 | 品牌编码，由 EZR 分配 |
| Sign | string | 是 | 请求签名 |
| Timestamp | string | 是 | 请求时间戳，格式 `yyyyMMddHHmmss` |
| AppSystem | string | 是 | 系统来源标识 |
| Args | string | 是 | 业务请求参数JSON字符串 |

### 业务请求参数（Args）

Args为JSON字符串，解析后包含以下字段：

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| OldCode | string | 是 | 会员线下卡号 |
| LockdedKey | string | 是 | 锁定流水号（来自积分锁定接口） |

### 请求示例

```json
{
    "AppId": "EZP-ZhuTest02",
    "Sign": "A1B2C3D4E5F6G7H8I9J0K1L2M3N4O5P6Q7R8S9T0",
    "Timestamp": "20260324143052",
    "AppSystem": "POS",
    "Args": "{\"OldCode\":\"C0000000739371B429\",\"LockdedKey\":\"42930577921776319594146948\"}"
}
```

## 完整示例代码

### 业务服务示例

```csharp
using System;
using System.Collections.Generic;
using EZP.Framework.Util.HttpClientFactory;
using EZR.OP.ThirdPartyProxy.Bussiness.OpenApi.Client;

/// <summary>
/// 积分解锁服务
/// </summary>
public static class UnlockBonusService
{
    private static readonly IEZROpenApiClient _openApiClient = EZP.Framework.Util.HttpClientFactory.EzpFrameworkDI.GetRequiredService<IEZROpenApiClient>();

    /// <summary>
    /// [EZR-API:1233] 解锁会员积分
    /// 解锁会员积分（接口编号：1233）
    /// </summary>
    /// <param name="request">请求参数</param>
    /// <returns>解锁结果</returns>
    public static BaseApiResInfo<UnlockBonusResult> UnlockBonus(UnlockBonusRequest request)
    {
        if (request == null)
        {
            return new BaseApiResInfo<UnlockBonusResult>
            {
                Status = false,
                StatusCode = 1001,
                Msg = "请求参数不能为空"
            };
        }

        if (string.IsNullOrWhiteSpace(request.OldCode))
        {
            return new BaseApiResInfo<UnlockBonusResult>
            {
                Status = false,
                StatusCode = 1002,
                Msg = "会员线下卡号不能为空"
            };
        }

        if (string.IsNullOrWhiteSpace(request.LockdedKey))
        {
            return new BaseApiResInfo<UnlockBonusResult>
            {
                Status = false,
                StatusCode = 1003,
                Msg = "锁定流水号不能为空"
            };
        }

        return _openApiClient.PostOpenApi<UnlockBonusRequest, UnlockBonusResult>("/api/cvip/unlockbonus", request);
    }
}

// 积分解锁请求参数（接口编号：1233）
public class UnlockBonusRequest
{
    /// <summary>
    /// 会员线下卡号
    /// </summary>
    public string OldCode { get; set; }

    /// <summary>
    /// 锁定流水号（来自积分锁定接口）
    /// </summary>
    public string LockdedKey { get; set; }
}

// 积分解锁响应结果（接口编号：1233）
public class UnlockBonusResult
{
    /// <summary>
    /// 结果状态码，200表示成功
    /// </summary>
    public int ResultCode { get; set; }

    /// <summary>
    /// 锁定流水号
    /// </summary>
    public string LockedKey { get; set; }

    /// <summary>
    /// 错误消息
    /// </summary>
    public string ErrorMsg { get; set; }

    /// <summary>
    /// 是否成功
    /// </summary>
    public bool Success { get; set; }

    /// <summary>
    /// 会员线下卡号
    /// </summary>
    public string OldCode { get; set; }

    /// <summary>
    /// 详细信息列表
    /// </summary>
    public List<object> Details { get; set; }
}

// 使用示例
class Program
{
    static void Main()
    {
        // 解锁积分请求
        var request = new UnlockBonusRequest
        {
            OldCode = "C0000000739371B429",
            LockdedKey = "42930577921776319594146948"
        };

        var result = UnlockBonusService.UnlockBonus(request);

        if (result.Status && result.Result.Success)
        {
            Console.WriteLine($"解锁成功，锁定流水号: {result.Result.LockedKey}");
            Console.WriteLine($"会员卡号: {result.Result.OldCode}");
            Console.WriteLine($"结果状态码: {result.Result.ResultCode}");
        }
        else
        {
            Console.WriteLine($"错误码: {result.StatusCode}, 错误信息: {result.Msg}");
            if (!string.IsNullOrWhiteSpace(result.Result?.ErrorMsg))
            {
                Console.WriteLine($"业务错误: {result.Result.ErrorMsg}");
            }
        }
    }
}
```
