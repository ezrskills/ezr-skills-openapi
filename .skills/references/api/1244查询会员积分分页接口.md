### 接口说明

#### 接口基础信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1244 |
| 接口名称 | 查询会员积分分页 |
| 接口地址 | /api/cvip/GetBonusPage |
| 协议 | HTTPS |
| 请求方式 | POST |
| 数据格式 | application/x-www-form-urlencoded |
| 字符编码 | UTF-8 |

#### 功能描述

用于分页查询会员积分流水记录，支持按时间范围、变动类型等条件筛选，适用于积分明细查询、积分流水展示等场景。

## 请求结构

### 外层请求参数

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| AppId | string | 是 | 品牌编码，由 EZR 分配 |
| Sign | string | 是 | 请求签名 |
| Timestamp | string | 是 | 请求时间戳，格式 `yyyyMMddHHmmss` |
| AppSystem | string | 是 | 系统来源标识 |
| Args | string | 是 | 业务请求参数JSON字符串 |

### 业务请求参数（Args）

Args为JSON字符串，解析后包含以下字段：

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| MobileNo | string | 否 | 手机号码 |
| OldCode | string | 否 | 会员线下卡号 |
| Code | string | 否 | 会员编码 |
| PageIndex | int | 否 | 页码，默认1 |
| PageSize | int | 否 | 每页数量，默认20，最大100 |

### 请求示例

```json
{
    "AppId": "EZP-ZhuTest02",
    "Sign": "A1B2C3D4E5F6G7H8I9J0K1L2M3N4O5P6Q7R8S9T0",
    "Timestamp": "20260324143052",
    "AppSystem": "CRM",
    "Args": "{\"MobileNo\":\"13850366666\",\"OldCode\":\"\",\"Code\":\"\",\"PageIndex\":1,\"PageSize\":10}"
}
```

```csharp
using System;
using System.Collections.Generic;
using EZP.Framework.Util.HttpClientFactory;
using EZR.OP.ThirdPartyProxy.Bussiness.OpenApi.Client;

/// <summary>
/// 会员积分服务
/// </summary>
public static class VipBonusService
{
    private static readonly IEZROpenApiClient _openApiClient = EZP.Framework.Util.HttpClientFactory.EzpFrameworkDI.GetRequiredService<IEZROpenApiClient>();

    /// <summary>
    /// [EZR-API:1244] 分页查询会员积分流水
    /// 分页查询会员积分流水（接口编号：1244）
    /// </summary>
    /// <param name="request">请求参数</param>
    /// <returns>积分分页数据</returns>
    public static BaseApiResInfo<List<BonusPageResult>> GetBonusPage(BonusPageRequest request)
    {
        if (request == null)
        {
            return new BaseApiResInfo<List<BonusPageResult>>
            {
                Status = false,
                StatusCode = 1001,
                Msg = "请求参数不能为空"
            };
        }

        return _openApiClient.PostOpenApi<BonusPageRequest, List<BonusPageResult>>("/api/cvip/GetBonusPage", request);
    }
}

// 查询会员积分分页请求参数（接口编号：1244）
public class BonusPageRequest
{
    /// <summary>
    /// 手机号码
    /// </summary>
    public string MobileNo { get; set; }

    /// <summary>
    /// 会员线下卡号
    /// </summary>
    public string OldCode { get; set; }

    /// <summary>
    /// 会员编码
    /// </summary>
    public string Code { get; set; }

    /// <summary>
    /// 页码，默认1
    /// </summary>
    public int PageIndex { get; set; } = 1;

    /// <summary>
    /// 每页数量，默认20，最大100
    /// </summary>
    public int PageSize { get; set; } = 20;
}

// 积分流水信息（接口编号：1244）
public class BonusPageResult
{
    /// <summary>
    /// 交易来源编码
    /// </summary>
    public int TransOriginInt { get; set; }

    /// <summary>
    /// 交易来源名称
    /// </summary>
    public string TransOrigin { get; set; }

    /// <summary>
    /// 流水号
    /// </summary>
    public string TransId { get; set; }

    /// <summary>
    /// 变动积分
    /// </summary>
    public int TransVal { get; set; }

    /// <summary>
    /// 交易日期
    /// </summary>
    public string TransDate { get; set; }

    /// <summary>
    /// 生效日期
    /// </summary>
    public string EffectDate { get; set; }

    /// <summary>
    /// 过期时间
    /// </summary>
    public string ExpiredExcTime { get; set; }

    /// <summary>
    /// 关联名称
    /// </summary>
    public string RelationName { get; set; }

    /// <summary>
    /// 关联编码
    /// </summary>
    public string RelationCode { get; set; }

    /// <summary>
    /// 变动后积分
    /// </summary>
    public int Bonus { get; set; }

    /// <summary>
    /// 总积分
    /// </summary>
    public int BonusTotal { get; set; }

    /// <summary>
    /// 总数量
    /// </summary>
    public int TotalCount { get; set; }

    /// <summary>
    /// 创建时间
    /// </summary>
    public string CreateDate { get; set; }
}

// 使用示例
class Program
{
    static void Main()
    {
        var request = new BonusPageRequest
        {
            MobileNo = "13850366666",
            OldCode = "",
            Code = "",
            PageIndex = 1,
            PageSize = 10
        };

        var result = VipBonusService.GetBonusPage(request);

        if (result.Status && result.Result != null)
        {
            foreach (var item in result.Result)
            {
                Console.WriteLine($"流水号: {item.TransId}");
                Console.WriteLine($"交易来源: {item.TransOrigin}");
                Console.WriteLine($"变动积分: {item.TransVal}");
                Console.WriteLine($"交易日期: {item.TransDate}");
                Console.WriteLine($"变动后积分: {item.Bonus}");
                Console.WriteLine($"总积分: {item.BonusTotal}");
                Console.WriteLine($"关联名称: {item.RelationName}");
                Console.WriteLine("---");
            }
        }
        else
        {
            Console.WriteLine($"错误码: {result.StatusCode}, 错误信息: {result.Msg}");
        }
    }
}
```
