### 接口说明

#### 接口基础信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1302 |
| 接口名称 | 在线核销券 |
| 接口地址 | /api/ccoup/CoupUse |
| 协议 | HTTPS |
| 请求方式 | POST |
| 数据格式 | application/x-www-form-urlencoded |
| 字符编码 | UTF-8 |

#### 功能描述

用于在线核销优惠券，支持单张或多张券同时核销。

## 请求结构

### 外层请求参数

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| AppId | string | 是 | 品牌编码，由 EZR 分配 |
| Sign | string | 是 | 请求签名 |
| Timestamp | string | 是 | 请求时间戳，格式 `yyyyMMddHHmmss` |
| AppSystem | string | 是 | 系统来源标识 |
| Args | string | 是 | 业务请求参数JSON字符串 |

### 业务请求参数（Args）

Args为JSON字符串，解析后包含以下字段：

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| ShopCode | string | 是 | 门店编码 |
| SalesNo | string | 是 | 销售单号 |
| SalesMoney | decimal | 是 | 销售金额 |
| CouponNos | array | 是 | 券编号列表 |
| ConsumeBrand | string | 否 | 核销品牌 |
| ConsumeUser | string | 否 | 核销人 |

### 请求示例

```json
{
    "AppId": "1001",
    "Sign": "A1B2C3D4E5F6G7H8I9J0K1L2M3N4O5P6Q7R8S9T0",
    "Timestamp": "20260323143052",
    "AppSystem": "POS",
    "Args": "{\"ShopCode\":\"HMGSTC\",\"SalesNo\":\"HMGSTC00001\",\"SalesMoney\":990,\"CouponNos\":[\"00000000000487567348\"],\"ConsumeBrand\":\"\",\"ConsumeUser\":\"\"}"
}
```

## 响应结构

### 标准响应格式

```json
{
    "Sign": "ECAA848AF880A5A61C6AC5DDF70A72C1ED0685B6",
    "Result": [
        {
            "CouponNo": "00000000000487567348",
            "Code": 200,
            "Message": "验证成功并消费",
            "CouponOrderPayAmount": 0
        }
    ],
    "Status": true,
    "StatusCode": 200,
    "Msg": "成功",
    "Timestamp": "20260416144057"
}
```

### 响应参数说明

| 参数名 | 类型 | 说明 |
|--------|------|------|
| Status | bool | 请求是否成功 |
| StatusCode | int | 状态码，200表示成功 |
| Msg | string | 响应消息 |
| Sign | string | 响应签名 |
| Timestamp | string | 响应时间戳 |
| Result | array | 核销结果列表 |

### 核销结果（Result项）

| 参数名 | 类型 | 说明 |
|--------|------|------|
| CouponNo | string | 券编号 |
| Code | int | 核销状态码，200表示成功 |
| Message | string | 核销消息 |
| CouponOrderPayAmount | decimal | 券抵扣金额 |

using System;
using System.Collections.Generic;
using EZP.Framework.Util.HttpClientFactory;
using EZR.OP.ThirdPartyProxy.Bussiness.OpenApi.Client;

/// <summary>
/// 券核销服务
/// </summary>
public static class CouponUseService
{
    private static readonly IEZROpenApiClient _openApiClient = EZP.Framework.Util.HttpClientFactory.EzpFrameworkDI.GetRequiredService<IEZROpenApiClient>();

    /// <summary>
    /// [EZR-API:1302] 在线核销券
    /// 在线核销券（接口编号：1302）
    /// </summary>
    /// <param name="request">请求参数</param>
    /// <returns></returns>
    public static BaseApiResInfo<List<CoupUseResult>> UseCoupon(CoupUseRequest request)
    {
        if (request == null)
        {
            return new BaseApiResInfo<List<CoupUseResult>>
            {
                Status = false,
                StatusCode = 1001,
                Msg = "请求参数不能为空"
            };
        }

        if (string.IsNullOrWhiteSpace(request.ShopCode))
        {
            return new BaseApiResInfo<List<CoupUseResult>>
            {
                Status = false,
                StatusCode = 1002,
                Msg = "门店编码不能为空"
            };
        }

        if (string.IsNullOrWhiteSpace(request.SalesNo))
        {
            return new BaseApiResInfo<List<CoupUseResult>>
            {
                Status = false,
                StatusCode = 1003,
                Msg = "销售单号不能为空"
            };
        }

        if (request.CouponNos == null || request.CouponNos.Count == 0)
        {
            return new BaseApiResInfo<List<CoupUseResult>>
            {
                Status = false,
                StatusCode = 1004,
                Msg = "券编号列表不能为空"
            };
        }

        return _openApiClient.PostOpenApi<CoupUseRequest, List<CoupUseResult>>("/api/ccoup/CoupUse", request);
    }
}

// 券核销请求参数（接口编号：1302）
public class CoupUseRequest

    /// 门店编码

class Program
{

        var result = CouponUseService.UseCoupon(request);

        if (result.Status && result.Result != null)

            foreach (var item in result.Result)

                Console.WriteLine($"券编号: {item.CouponNo}");
                Console.WriteLine($"核销状态码: {item.Code}");
                Console.WriteLine($"核销消息: {item.Message}");
                Console.WriteLine($"券抵扣金额: {item.CouponOrderPayAmount}");
            }
        }
        else
        {
            Console.WriteLine($"错误码: {result.StatusCode}, 错误信息: {result.Msg}");
        }
    }
}
