### 接口说明

#### 接口基础信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1227 |
| 接口名称 | 获取会员标签信息 |
| 接口地址 | `/api/cvip/GetTags` |
| 协议 | HTTPS |
| 请求方式 | POST |
| 数据格式 | application/x-www-form-urlencoded |
| 字符编码 | UTF-8 |


#### 功能描述

用于查询会员已绑定的标签信息，支持标签查询、分组过滤，返回标签分组名称、标签ID和标签名称。

## 请求结构

### 外层请求参数

所有接口的外层请求参数结构一致：

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| AppId | string | 是 | 品牌编码，由 EZR 分配 |
| Sign | string | 是 | 请求签名，参见签名规则 |
| Timestamp | string | 是 | 请求时间戳，格式 `yyyyMMddHHmmss` |
| AppSystem | string | 是 | 系统来源标识 |
| Args | string | 是 | 业务请求参数JSON字符串，结构因接口而异 |


| 参数名 | 类型 | 说明 |

| Timestamp | string | 响应时间戳 |

### 响应签名验证

响应签名生成规则与请求签名相同：
```
Sign = SHA1("AppId={AppId}&Timestamp={Timestamp}&Token={Token}")
```

## 接口详情

### 获取会员标签信息（1227）

#### 接口基本信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1227 |
| 接口名称 | 获取会员标签信息 |
| 接口地址 | `/api/cvip/GetTags` |
| 请求方式 | POST |
| 功能描述 | 查询会员已绑定的标签信息 |

#### 业务请求参数（Args）

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| OldCode | string | 否* | 会员线下卡号（最多80位） |
| Code | string | 否* | 会员线上卡号（最多20位） |
| MobileNo | string | 否* | 手机号码 |
| WxOpenID | string | 否* | 微信 OpenID |
| GroupIds | array[int] | 否 | 标签分组ID列表，用于过滤指定分组的标签 |

> *以上参数（OldCode、Code、MobileNo、WxOpenID）至少填写一个

#### 业务响应参数（Result）

| 参数名 | 类型 | 说明 |
|--------|------|------|
| GroupName | string | 标签分组名称 |
| GroupId | int | 标签分组ID |
| Tags | array[string] | 标签名称列表 |
| TagIds | array[int] | 标签ID列表 |

```csharp
using System;
using System.Collections.Generic;
using EZP.Framework.Util.HttpClientFactory;
using EZR.OP.ThirdPartyProxy.Bussiness.OpenApi.Client;

/// <summary>
/// 会员服务
/// </summary>
public static class VipService
{
    private static readonly IEZROpenApiClient _openApiClient = EZP.Framework.Util.HttpClientFactory.EzpFrameworkDI.GetRequiredService<IEZROpenApiClient>();

    /// <summary>
    /// [EZR-API:1227] 获取会员标签信息
    /// 获取会员标签信息
    /// </summary>
    /// <param name="request">请求参数</param>
    /// <returns></returns>
    public static BaseApiResInfo<List<VipTagItemResult>> GetVipTags(VipTagsGetRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.OldCode) &&
            string.IsNullOrWhiteSpace(request.Code) &&
            string.IsNullOrWhiteSpace(request.MobileNo) &&
            string.IsNullOrWhiteSpace(request.WxOpenID))
        {
            return new BaseApiResInfo<List<VipTagItemResult>>
            {
                Status = false,
                StatusCode = 1001,
                Msg = "会员线下卡号/线上卡号/手机号/微信OpenID至少填写一个"
            };
        }

        return _openApiClient.PostOpenApi<VipTagsGetRequest, List<VipTagItemResult>>("/api/cvip/GetTags", request);
    }
}

// 获取会员标签请求参数（接口编号：1227）
public class VipTagsGetRequest
{
    /// <summary>
    /// 会员线下卡号（最多80位）

    /// </summary>
    public string Code { get; set; }

    /// 手机号码

    public string MobileNo { get; set; }

    /// <summary>
    /// 微信 OpenID
    /// </summary>
    public string WxOpenID { get; set; }

    /// <summary>
    /// 标签分组ID列表（用于过滤指定分组的标签）
    /// </summary>
    public List<int> GroupIds { get; set; }

public class VipTagItemResult

    /// <summary>
    /// 标签分组名称

    /// <summary>

    /// </summary>
    public int GroupId { get; set; }

// 使用示例
class Program
{
    static void Main()
    {
        // 通过线下卡号查询所有标签
        var requestByOldCode = new VipTagsGetRequest { OldCode = "EZ00000277735638" };
        var resultByOldCode = VipService.GetVipTags(requestByOldCode);

        if (resultByOldCode.Status)
        {
            Console.WriteLine($"会员标签分组数: {resultByOldCode.Result.Count}");
            foreach (var group in resultByOldCode.Result)
            {
                Console.WriteLine($"分组: {group.GroupName} (ID: {group.GroupId})");
                Console.WriteLine($"标签: {string.Join(", ", group.Tags)}");
                Console.WriteLine($"标签ID: {string.Join(", ", group.TagIds)}");
            }
        }
        else
        {
            Console.WriteLine($"错误码: {resultByOldCode.StatusCode}, 错误信息: {resultByOldCode.Msg}");
        }

        // 通过手机号查询
        var requestByMobile = new VipTagsGetRequest { MobileNo = "13800138000" };
        var resultByMobile = VipService.GetVipTags(requestByMobile);

        if (resultByMobile.Status)
        {
            Console.WriteLine($"手机号查询成功，分组数: {resultByMobile.Result.Count}");
        }
        else
        {
            Console.WriteLine($"错误码: {resultByMobile.StatusCode}, 错误信息: {resultByMobile.Msg}");
        }

        // 按分组ID过滤查询
        var filterRequest = new VipTagsGetRequest
        {
            OldCode = "EZ00000277735638",
            GroupIds = new List<int> { 10010 }
        };
        var filterResult = VipService.GetVipTags(filterRequest);

        if (filterResult.Status)
        {
            Console.WriteLine($"按分组ID过滤的标签:");
            foreach (var group in filterResult.Result)
            {
                Console.WriteLine($"分组: {group.GroupName}");
                foreach (var tag in group.Tags)
                {
                    Console.WriteLine($"- {tag}");
                }
            }
        }
    }
}
```
