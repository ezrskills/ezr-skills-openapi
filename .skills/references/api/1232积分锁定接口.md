### 接口说明

#### 接口基础信息

| 项目 | 说明 |
|------|------|
| 接口编号 | 1232 |
| 接口名称 | 积分锁定 |
| 接口地址 | /api/cvip/lockbonus |
| 协议 | HTTPS |
| 请求方式 | POST |
| 数据格式 | application/x-www-form-urlencoded |
| 字符编码 | UTF-8 |

#### 功能描述

用于锁定会员积分，锁定后积分不可使用但保留，适用于积分兑换预占、积分抵扣预占等场景。

## 请求结构

### 外层请求参数

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| AppId | string | 是 | 品牌编码，由 EZR 分配 |
| Sign | string | 是 | 请求签名 |
| Timestamp | string | 是 | 请求时间戳，格式 `yyyyMMddHHmmss` |
| AppSystem | string | 是 | 系统来源标识 |
| Args | string | 是 | 业务请求参数JSON字符串 |

### 业务请求参数（Args）

Args为JSON字符串，解析后包含以下字段：

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| OldCode | string | 是 | 会员线下卡号 |
| Bonus | int | 是 | 锁定积分数量 |
| Remark | string | 否 | 备注 |
| unCkBonus | bool | 否 | 是否未扣减积分，默认false |

### 请求示例

```json
{
    "AppId": "EZP-ZhuTest02",
    "Sign": "A1B2C3D4E5F6G7H8I9J0K1L2M3N4O5P6Q7R8S9T0",
    "Timestamp": "20260324143052",
    "AppSystem": "POS",
    "Args": "{\"OldCode\":\"C0000000739371B429\",\"Bonus\":100,\"Remark\":\"\",\"unCkBonus\":false}"
}
```

## 完整示例代码

### 业务服务示例

```csharp
using System;
using System.Collections.Generic;
using EZP.Framework.Util.HttpClientFactory;
using EZR.OP.ThirdPartyProxy.Bussiness.OpenApi.Client;

/// <summary>
/// 积分锁定服务
/// </summary>
public static class LockBonusService
{
    private static readonly IEZROpenApiClient _openApiClient = EZP.Framework.Util.HttpClientFactory.EzpFrameworkDI.GetRequiredService<IEZROpenApiClient>();

    /// <summary>
    /// [EZR-API:1232] 锁定会员积分
    /// 锁定会员积分（接口编号：1232）
    /// </summary>
    /// <param name="request">请求参数</param>
    /// <returns>锁定结果</returns>
    public static BaseApiResInfo<LockBonusResult> LockBonus(LockBonusRequest request)
    {
        if (request == null)
        {
            return new BaseApiResInfo<LockBonusResult>
            {
                Status = false,
                StatusCode = 1001,
                Msg = "请求参数不能为空"
            };
        }

        if (string.IsNullOrWhiteSpace(request.OldCode))
        {
            return new BaseApiResInfo<LockBonusResult>
            {
                Status = false,
                StatusCode = 1002,
                Msg = "会员线下卡号不能为空"
            };
        }

        if (request.Bonus <= 0)
        {
            return new BaseApiResInfo<LockBonusResult>
            {
                Status = false,
                StatusCode = 1003,
                Msg = "锁定积分数量必须大于0"
            };
        }

        return _openApiClient.PostOpenApi<LockBonusRequest, LockBonusResult>("/api/cvip/lockbonus", request);
    }
}

// 积分锁定请求参数（接口编号：1232）
public class LockBonusRequest
{
    /// <summary>
    /// 会员线下卡号
    /// </summary>
    public string OldCode { get; set; }

    /// <summary>
    /// 锁定积分数量
    /// </summary>
    public int Bonus { get; set; }

    /// <summary>
    /// 备注
    /// </summary>
    public string Remark { get; set; }

    /// <summary>
    /// 是否未扣减积分，默认false
    /// </summary>
    public bool unCkBonus { get; set; }
}

// 积分锁定响应结果（接口编号：1232）
public class LockBonusResult
{
    /// <summary>
    /// 结果状态码，200表示成功
    /// </summary>
    public int ResultCode { get; set; }

    /// <summary>
    /// 锁定流水号，用于后续消费或解锁
    /// </summary>
    public string LockedKey { get; set; }

    /// <summary>
    /// 错误消息
    /// </summary>
    public string ErrorMsg { get; set; }

    /// <summary>
    /// 是否成功
    /// </summary>
    public bool Success { get; set; }

    /// <summary>
    /// 会员线下卡号
    /// </summary>
    public string OldCode { get; set; }

    /// <summary>
    /// 详细信息列表
    /// </summary>
    public List<object> Details { get; set; }
}

// 使用示例
class Program
{
    static void Main()
    {
        // 锁定积分请求
        var request = new LockBonusRequest
        {
            OldCode = "C0000000739371B429",
            Bonus = 100,
            Remark = "",
            unCkBonus = false
        };

        var result = LockBonusService.LockBonus(request);

        if (result.Status && result.Result.Success)
        {
            Console.WriteLine($"锁定成功，锁定流水号: {result.Result.LockedKey}");
            Console.WriteLine($"会员卡号: {result.Result.OldCode}");
            Console.WriteLine($"结果状态码: {result.Result.ResultCode}");
        }
        else
        {
            Console.WriteLine($"错误码: {result.StatusCode}, 错误信息: {result.Msg}");
            if (!string.IsNullOrWhiteSpace(result.Result?.ErrorMsg))
            {
                Console.WriteLine($"业务错误: {result.Result.ErrorMsg}");
            }
        }
    }
}
```
